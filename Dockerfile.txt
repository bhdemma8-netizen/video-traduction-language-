FROM python:3.11-slim

# ffmpeg est indispensable pour extraire/fusionner l'audio des vidéos
RUN apt-get update && apt-get install -y --no-install-recommends ffmpeg \
    && rm -rf /var/lib/apt/lists/*

WORKDIR /app

COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

COPY app.py .

# Timeout élevé car le traitement vidéo prend du temps ; 1 worker car le
# traitement se fait déjà en thread de fond (pattern job_id + polling)
CMD ["sh", "-c", "gunicorn -w 1 -b 0.0.0.0:$PORT --timeout 900 app:app"]
